import streamlit as st
import pickle
import pandas as pd

# ✅ Load data ONCE at the top (not inside functions)
movies_dict = pickle.load(open('movies.pkl', 'rb'))
new_df = pd.DataFrame(movies_dict)
similarity = pickle.load(open('similarity.pkl', 'rb'))

# ✅ Create lowercase column ONCE outside the function
new_df['title_lower'] = new_df['title'].str.lower()

# ✅ Recommend function
def recommend(movie):
    movie = movie.lower()

    if movie not in new_df['title_lower'].values:
        return []

    movie_index = new_df[new_df['title_lower'] == movie].index[0]
    distances = similarity[movie_index]
    movies_list = sorted(list(enumerate(distances)), key=lambda x: x[1], reverse=True)

    recommended_movies = []
    for i in movies_list[1:6]:
        recommended_movies.append(new_df.iloc[i[0]].title)

    return recommended_movies

# ✅ Streamlit UI
st.title('🎬 Movie Recommendation System')

selected_movie = st.selectbox('Select a movie:', new_df['title'].values)

if st.button('Recommend'):
    recommendations = recommend(selected_movie)
    if recommendations:
        st.subheader('Top 5 Recommendations:')
        for movie in recommendations:
            st.write(f"🎥 {movie}")
    else:
        st.error("Movie not found!")
